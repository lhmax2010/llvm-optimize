315: class ELF:
316:     """ELF64 little-endian reader. No whole-file loads; semantic symbol/relocation records."""
317:     def __init__(self, path):
318:         self.path = Path(path); self.f = self.path.open('rb')
319:         if os.fstat(self.f.fileno()).st_size < 64:
320:             self.f.close(); raise GateError('truncated ELF header')

322:         try:
323:             require(self.mm[:6] == b'\x7fELF\x02\x01', 'expected ELF64 little endian')
324:             h = self.unpack('<16sHHIQQQIHHHHHH', 0)
325:             require(h[0][6] == 1 and h[3] == 1 and h[8] == 64, 'invalid ELF header version/size')
326:             require(h[1] in (2, 3), 'expected executable or shared ELF type')
327:             self.elf_type = h[1]; self.machine = h[2]; self.entry = h[4]
328:             phoff, shoff, phentsize, phnum, shentsize, shnum, stridx = h[5], h[6], h[9], h[10], h[11], h[12], h[13]
329:             require(shentsize == 64 and shnum > 0 and stridx < shnum, 'unsupported section numbering')
330:             self.sections = [self.unpack('<IIQQQQIIQQ', shoff + i * 64) for i in range(shnum)]
331:             for section in self.sections:
332:                 if section[1] != 8:
333:                     self.range(section[4], section[5])
334:             names = self.sections[stridx]
335:             require(names[1] == 3, 'section name table is not STRTAB')
336:             self.names = [self.string(names[4], names[5], s[0]) for s in self.sections]
337:             self.by_name = dict(zip(self.names, self.sections))
338:             require(len(self.by_name) == len(self.names), 'duplicate section name')
339:             self.loads = []; self.phdrs = []

994: def disk_entries(root):
995:     """lstat allocation, including directories; never follow symlinks."""
996:     root = Path(root)
997:     if not root.exists() and not root.is_symlink():
998:         return {}
999:     result = {}
1000:     def add(path):
1001:         s = path.lstat()
1002:         result[path.relative_to(root).as_posix()] = {
1003:             'device': s.st_dev, 'inode': s.st_ino, 'bytes': s.st_size,
1004:             'allocated_bytes': s.st_blocks*512, 'mode': s.st_mode}
1005:     add(root)
1006:     if root.is_dir() and not root.is_symlink():
1007:         def failed(error): raise error
1008:         for directory, dirs, files in os.walk(root, followlinks=False, onerror=failed):
1009:             for name in dirs+files:
1010:                 add(Path(directory)/name)
1011:     return result

1465: def package_compare(ordinary, candidate, mode, proof):
1466:     """Local RPM payload comparison. Proof binds per-path DWARF provenance to baseline.
1467: 
1468:     The proof must be generated/reviewed from ordinary DWARF and ALL other debuginfo
1469:     (see design). Missing evidence fails; a path pattern alone is never proof.
1470:     Root entries are package-qualified: main/, debuginfo/, debugsource/.
1471:     """
1472:     require(mode in ('debug','release'), 'unknown package mode')
1473:     old, new = ordinary['files'], candidate['files']
1474:     require(new.keys() <= old.keys(), 'RPM added a file')
1475:     entity = 'main/usr/bin/clang-22'
1476:     require(entity in old and entity in new, 'RPM lost clang-22')
1477:     require(old[entity]['type'] == new[entity]['type'] == 'file' and
1478:             old[entity]['mode'] == new[entity]['mode'], 'clang entity type/mode changed')
1479:     aliases = set(proof.get('aliases', []))
1480:     for key in aliases:
1481:         require(key.startswith('main/usr/bin/') and key != entity and key in old,
1482:                 'invalid baseline alias')
1483:         # Alias evidence is produced by alias_group on the ordinary extraction root.
1484:         require(proof.get('alias_baseline', {}).get(key) == old[key], 'unproven clang alias')
1485:     removed = sorted(old.keys()-new.keys()); changed = []
1486:     for key in sorted(new):
1487:         if old[key] == new[key]:
1488:             continue
1489:         if key == entity:
1490:             changed.append(key);continue
1491:         if key in aliases:
1492:             require(new[key] == {'type':'symlink','target':'clang-22'}, 'alias is not normalized')
1493:             changed.append(key);continue
1494:         raise GateError('non-clang RPM file changed: '+key)
1495:     for key in removed:
1496:         require(mode == 'debug', 'release RPM lost a file: '+key)
1497:         item = proof.get('removed', {}).get(key)
1498:         require(item is not None and item.get('baseline') == old[key], 'missing or mismatched DWARF provenance: '+key)
1499:         evidence = item.get('evidence', {})
1500:         require(evidence.get('clang_debug_sha256') and evidence.get('dwarf_dump_sha256') and
1501:                 evidence.get('all_other_debug_manifest_sha256'), 'incomplete DWARF evidence: '+key)
1502:         if key.startswith('debuginfo/'):
1503:             require(item.get('kind') in ('clang_debug','clang_build_id') and
1504:                     evidence.get('owner') == 'usr/bin/clang-22', 'debug file not owned by clang-22')
1505:         elif key.startswith('debugsource/'):
1506:             require(item.get('kind') == 'exclusive_source' and evidence.get('referenced_by') == ['usr/bin/clang-22'] and
1507:                     evidence.get('source_path'), 'source not proven exclusive to clang-22')
1508:         else:
1509:             raise GateError('unexpected missing file: '+key)
1510:     return {'mode':mode, 'pass':True, 'changed':changed, 'removed':removed,
1511:             'proof_sha256':hashlib.sha256(json.dumps(proof,sort_keys=True).encode()).hexdigest()}
