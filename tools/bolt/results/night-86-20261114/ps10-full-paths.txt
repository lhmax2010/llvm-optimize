347: def pic_relocations(data):
348:     """Reject absolute relocations in read-only allocated x86_64 sections.
349: 
350:     Debug relocations are excluded. Absolute 32-bit relocations against
351:     non-absolute symbols are also rejected in writable allocated sections.
352:     PC-relative relocations are not all forbidden: e.g. hidden symbols and
353:     .eh_frame legitimately use them. A consumer GNU ld -shared check follows.
354:     """
355:     if member_kind(data) != 'machine' or data[4:6] != b'\x02\x01' or struct.unpack_from('<H', data, 18)[0] != 62:
356:         raise ValueError('not x86_64 ELF64 little-endian ET_REL')
357:     off = struct.unpack_from('<Q', data, 40)[0]
358:     stride, count = struct.unpack_from('<HH', data, 58)
359:     if not count:
360:         count = struct.unpack_from('<IIQQQQIIQQ', data, off)[5]
361:     sections = [struct.unpack_from('<IIQQQQIIQQ', data, off+i*stride) for i in range(count)]
362:     checked, forbidden = 0, []
363:     for section in sections:
364:         if section[1] not in (4, 9):
365:             continue
366:         target = sections[section[7]]
367:         if not target[2] & 2:  # SHF_ALLOC; excludes all debug sections
368:             continue
369:         symbols = sections[section[6]]
370:         for pos in range(section[4], section[4]+section[5], section[9]):
371:             _, info = struct.unpack_from('<QQ', data, pos)
372:             kind, symbol_index = info & 0xffffffff, info >> 32
373:             symbol = struct.unpack_from('<IBBHQQ', data, symbols[4]+symbol_index*symbols[9])
374:             checked += 1
375:             # SHN_ABS references do not need runtime relocation.
376:             if symbol[3] != 0xfff1 and (kind in (10, 11) or (kind == 1 and not target[2] & 1)):
377:                 forbidden.append(dict(type=kind, target_section=section[7], symbol_index=symbol_index))
378:     return dict(checked=checked, forbidden=forbidden)
