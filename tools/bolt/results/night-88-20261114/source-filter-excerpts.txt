2868:                                        const RelocationRef &Rel) {
2869:   const bool IsAArch64 = BC->isAArch64();

2873:

2875:   Rel.getTypeName(TypeName);
2876:   uint32_t RType = Relocation::getType(Rel);
2877:   if (Relocation::skipRelocationType(RType))
2878:     return;
2879: 
2880:   // Adjust the relocation type as the linker might have skewed it.
2881:   if (IsX86 && (RType & ELF::R_X86_64_converted_reloc_bit)) {
2882:     if (opts::Verbosity >= 1)
2883:       dbgs() << "BOLT-WARNING: ignoring R_X86_64_converted_reloc_bit\n";
2884:     RType &= ~ELF::R_X86_64_converted_reloc_bit;
2885:   }
2886: 
2887:   if (Relocation::isTLS(RType)) {
2888:     // No special handling required for TLS relocations on X86.
2889:     if (IsX86)
2890:       return;
2891: 
2892:     // The non-got related TLS relocations on AArch64 and RISC-V also could be
2893:     // skipped.
2894:     if (!Relocation::isGOT(RType))
2895:       return;
2896:   }
2897: 
2898:   if (!IsAArch64 && BC->getDynamicRelocationAt(Rel.getOffset())) {
2899:     LLVM_DEBUG({

2901:              << "dynamic relocation against it. Ignoring static relocation.\n";
2902:     });
2903:     return;
2904:   }
2905: 
2906:   std::string SymbolName;
2907:   uint64_t SymbolAddress;
2908:   int64_t Addend;
2909:   uint64_t ExtractedValue;

2912:                          SymbolAddress, Addend, ExtractedValue)) {
