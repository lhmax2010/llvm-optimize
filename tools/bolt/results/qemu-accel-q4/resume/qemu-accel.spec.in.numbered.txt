331: prepare_package_list() {
332:   local bins=""
333:   local name=$1
334:   shift
335:   local executables="$@"
336:   echo "Setting up $name for $executables"
337:   ## Prepare the clang binaries separately
338:   for executable in $executables; do
339:     bins="$bins $executable `ldd $executable 2>/dev/null | sed -n 's,.*=> \(/[^ ]*\) .*,\1,p'`"
340:   done
341: 
342:   for b in $(echo "$bins" | tr ' ' '\n' | sort -u); do
343:     local orig_file=$(readlink -f $b)
344:     bins="$bins $orig_file"
345:     if ! echo "$b" | grep -q '/\.\./'; then
346:         # Ignore relative paths
347:         echo $b >> $name.list
348:     fi
349:     echo $orig_file >> $name.list
350:   done
351: 
352:   ## Store the resulting list for packaging, setting up the right prefix, removing duplication of %emul_path if one exists
353:   cat $name.list | \
354:       sed -e 's|^|%{emul_path}|;s| |\n%{emul_path}|g' | \
355:       grep -v "%{emul_path}%{emul_path}" | \
356:       sort -u > %{_sourcedir}/$name-accel.list
357:   binaries=$(echo "$binaries $bins" | tr ' ' '\n' | sort -u)
358: }

392: patch_binary() {
393:   local binary=$1
394:   local rtld=$2
395:   # We don't need to accelerate shell scripts
396:   local filetype=`file --brief $binary`
397:   if echo "$filetype" | grep -q '^POSIX shell script'; then
398:     return
399:   fi
400: 
401:   local outfile=${buildroot_emul_path}$binary
402:   if ! test -e "$binary"; then
403:     echo "WARNING: File not found, ignoring"
404:     return
405:   fi
406:   if test -f "$outfile"; then
407:     echo "WARNING: File exists, ignoring"
408:     return
409:   fi
410:   mkdir -p ${outfile%/*}
411:   cp -aL $binary $outfile
412:   [ "$binary" = "$rtld" ] && return
413:   [ "$binary" = "%{emul_path}$rtld" ] && return
414: 
415:   # not all binaries have an .interp section
416:   if patchelf --print-interpreter $outfile 1>/dev/null 2>/dev/null; then
417:     patchelf --set-interpreter %{emul_path}$rtld $outfile
418:   fi
419: 
420:   # Only patch RUNPATH if one exists and is absolute
421:   local rpath=$(readelf -d "$outfile" | sed -e '/RUNPATH/{s~.*\[\(.*\)\]~\1~;n};d')
422:   if [ -n "$rpath" ]; then
423:     first_char=$(echo "$rpath" | cut -c1)
424:     if [ "$first_char" = "/" ]; then
425:       patchelf --set-rpath "%{emul_path}%{_libdir}" $outfile
426:     fi
427:   fi
428: }
429: 
430: # Process all binaries directly (parallel jobs with simple file-based coordination)
431: for binary in $binaries; do
432:     patch_binary "$binary" "$LD" &
433: done
434: wait

439: # uname -a returns the host architecture instead of the target arch.
440: rm -f "${buildroot_emul_path}%{_bindir}/toybox"
441: sed -i "\|^%{emul_path}%{_bindir}/toybox$|d" %{_sourcedir}/all-accel.list 2>/dev/null
